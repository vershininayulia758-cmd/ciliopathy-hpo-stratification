# 7. Функции парсинга GeneReviews (без изменений)

def clean_text(text):
    text = re.sub(r'<[^>]+>', ' ', text)
    text = re.sub(r'\s+', ' ', text).strip()
    return text

def extract_clinical_section(content):
    text = clean_text(content)
    start_patterns = [
        r'Clinical\s+Characteristics',
        r'Clinical\s+Manifestations',
        r'Clinical\s+Features',
        r'Clinical\s+Description'
    ]
    start_pos = -1
    for pat in start_patterns:
        m = re.search(pat, text, re.IGNORECASE)
        if m:
            start_pos = m.end()
            break
    if start_pos == -1:
        return ""
    
    end_markers = [
        'Diagnosis', 'Management', 'Genetic Counseling',
        'Molecular Genetics', 'References', 'Summary',
        'Differential Diagnosis', 'Prevalence'
    ]
    end_pos = len(text)
    for marker in end_markers:
        idx = text.find(marker, start_pos)
        if idx != -1:
            end_pos = min(end_pos, idx)
    return text[start_pos:end_pos].strip()

def extract_frequency_sentences(section_text):
    sentences = re.split(r'(?<=[.!?])\s+', section_text)
    freq_patterns = [
        r'\d+\s*[-–]\s*\d+\s*%',
        r'\d+\s*%',
        r'>\s*\d+\s*%',
        r'<\s*\d+\s*%',
        r'~\s*\d+\s*%',
        r'\bmost\b', r'\bmajority\b', r'\bvirtually\s+all\b',
        r'\bcore\s+feature\b', r'\brare\b', r'\boccasional\b',
        r'\ball\s+individuals\b', r'\btypically\b', r'\bcommonly\b'
    ]
    freq_sent = []
    for sent in sentences:
        sent = sent.strip()
        if len(sent) < 10:
            continue
        if any(re.search(p, sent, re.IGNORECASE) for p in freq_patterns):
            freq_sent.append(sent)
    return freq_sent

def extract_percent(sent):
    m = re.search(r'(\d+)\s*%', sent)
    return int(m.group(1)) if m else None

def extract_freq_markers(sent):
    markers = []
    if re.search(r'\bmost\b|\bmajority\b|\bvirtually all\b|\ball individuals\b|\bcore feature\b', sent, re.I):
        markers.append('HIGH')
    if re.search(r'\brare\b|\boccasional\b|\bvery rare\b', sent, re.I):
        markers.append('LOW')
    if re.search(r'\btypically\b|\bcommonly\b', sent, re.I):
        markers.append('MODERATE')
    return ', '.join(markers) if markers else ''

def classify_system(hpo_term, text, category_from_excel=''):
    """Определяет систему органов. Если передана категория из Excel, использует её приоритетно."""
    if category_from_excel:
        return category_from_excel.lower()
    
    search = (hpo_term + ' ' + text).lower()
    system_keywords = {
        'neurological': ['brain', 'seizure', 'epilepsy', 'ataxia', 'cerebellar', 'cognitive', 'intellectual', 'mental retardation', 'developmental delay', 'hypotonia', 'spastic', 'neuro', 'encephalopathy', 'dystonia', 'tremor'],
        'ocular': ['eye', 'vision', 'retina', 'retinal', 'cataract', 'glaucoma', 'blind', 'optic', 'nystagmus', 'strabismus', 'cornea', 'lens', 'fundus', 'photophobia', 'microphthalmia', 'coloboma', 'night blindness'],
        'renal': ['kidney', 'renal', 'nephron', 'cyst', 'glomerulo', 'ureter', 'bladder', 'nephropathy', 'polyuria', 'proteinuria', 'hematuria', 'creatinine', 'dialysis', 'nephronophthisis'],
        'cardiac': ['heart', 'cardiac', 'cardiomyopathy', 'arrhythmia', 'valve', 'septal', 'hypertrophy', 'palpitations', 'syncope', 'myocardium', 'conduction', 'aortic'],
        'hepatic': ['liver', 'hepatic', 'hepat', 'cirrhosis', 'fibrosis', 'bile', 'jaundice', 'portal', 'transaminase', 'cholestasis', 'biliary'],
        'pulmonary': ['lung', 'pulmonary', 'respiratory', 'breath', 'asthma', 'copd', 'pneumonia', 'trachea', 'bronchi', 'emphysema', 'cough', 'dyspnea'],
        'skeletal': ['bone', 'skeleton', 'scoliosis', 'joint', 'arthritis', 'osteoporosis', 'fracture', 'short stature', 'limb', 'digit', 'polydactyly', 'syndactyly', 'cleft', 'cranio', 'brachydactyly'],
        'muscular': ['muscle', 'myopathy', 'dystrophy', 'weakness', 'myalgia', 'atrophy', 'hypertrophy', 'creatine kinase', 'emg'],
        'endocrine': ['thyroid', 'diabetes', 'insulin', 'hormone', 'pituitary', 'adrenal', 'growth', 'puberty', 'gonad', 'menstrual', 'obesity', 'metabolic', 'glucose'],
        'dermatological': ['skin', 'rash', 'pigment', 'hair', 'nail', 'ichthyosis', 'erythema', 'vesicle', 'ulcer', 'scar', 'hyperkeratosis', 'alopecia', 'collodion'],
        'immunological': ['immune', 'infection', 'autoimmune', 'allergy', 'antibody', 'lymphocyte', 'immuno', 'inflammation', 'sepsis'],
        'ear': ['hearing', 'deafness', 'auditory', 'ear', 'vestibular', 'cochlea', 'otosclerosis', 'tinnitus']
    }
    scores = Counter()
    for system, kws in system_keywords.items():
        for kw in kws:
            if kw in search:
                scores[system] += 1
    if scores:
        return scores.most_common(1)[0][0]
    return 'other'

# 8. Обработка GeneReviews для наших генов

gr_by_gene = defaultdict(list)

if not gr_df.empty:
    for idx, row in tqdm(gr_df.iterrows(), total=len(gr_df), desc="Парсинг GeneReviews"):
        content = row.get('content', '')
        if not content:
            continue
        keywords = row.get('keywords', '')
        title = row.get('title', '')
        if isinstance(keywords, list):
            keywords_str = ' '.join(keywords)
        else:
            keywords_str = str(keywords)
        if isinstance(title, list):
            title_str = ' '.join(title)
        else:
            title_str = str(title)
        
        matched_genes = set()
        for gene in GENES:
            gene_upper = gene.upper()
            if gene_upper in keywords_str.upper() or gene_upper in title_str.upper():
                matched_genes.add(gene_upper)
        
        if not matched_genes:
            continue
        
        section = extract_clinical_section(content)
        if not section:
            continue
        
        freq_sents = extract_frequency_sentences(section)
        for sent in freq_sents:
            percent = extract_percent(sent)
            markers = extract_freq_markers(sent)
            system = classify_system('', sent)
            for gene in matched_genes:
                gr_by_gene[gene].append({
                    'gene': gene,
                    'hpo_id': '',
                    'hpo_term': '',
                    'phenotype_text': sent,
                    'frequency': '',
                    'percent': percent,
                    'frequency_markers': markers,
                    'source': 'GeneReviews',
                    'system': system
                })
    print(f"Генов с GeneReviews описаниями: {len(gr_by_gene)}")
