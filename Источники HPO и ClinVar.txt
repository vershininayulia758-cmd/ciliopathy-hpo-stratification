# ==================== СТРУКТУРЫ ДАННЫХ ====================
@dataclass
class HPOData:
    hpo_id: str
    hpo_name: str
    gene_symbol: str
    frequency: str = ""
    evidence: str = ""
    disease: str = ""
    
    @property
    def frequency_score(self) -> float:
        freq_map = {
            'Obligate': 3.0,
            'Very frequent': 2.5,
            'Frequent': 2.0,
            'Occasional': 1.0,
            'Rare': 0.5,
            'Very rare': 0.2,
            'Excluded': 0.0
        }
        return freq_map.get(self.frequency, 1.0)

@dataclass
class ClinVarVariant:
    variant_id: str
    gene_symbol: str
    clinical_significance: str
    variant_type: str          # missense, nonsense, frameshift, splice, deletion, insertion, duplication, indel, synonymous, stop_loss, start_loss, intronic, utr, regulatory, intergenic, other
    review_status: str
    chromosome: str = ""
    position: str = ""
    age_of_onset: str = ""
    phenotype_ids: str = ""
    
    @property
    def is_pathogenic(self) -> bool:
        if not self.clinical_significance:
            return False
        sig = self.clinical_significance.lower()
        return ('pathogenic' in sig and 'likely' not in sig and 
                'benign' not in sig and 'conflict' not in sig and 'uncertain' not in sig)
    
    @property
    def is_likely_pathogenic(self) -> bool:
        if not self.clinical_significance:
            return False
        sig = self.clinical_significance.lower()
        return 'likely pathogenic' in sig or 'likely_pathogenic' in sig
    
    @property
    def is_vus(self) -> bool:
        if not self.clinical_significance:
            return False
        sig = self.clinical_significance.lower()
        return ('uncertain' in sig or 'vus' in sig or 'conflict' in sig or
                'not provided' in sig)
    
    @property
    def is_likely_benign(self) -> bool:
        if not self.clinical_significance:
            return False
        sig = self.clinical_significance.lower()
        return 'likely benign' in sig or 'likely_benign' in sig
    
    @property
    def is_benign(self) -> bool:
        if not self.clinical_significance:
            return False
        sig = self.clinical_significance.lower()
        return 'benign' in sig and 'likely' not in sig

    data_quality: float = 0.0

    def to_dict(self) -> Dict:
        return asdict(self)

# ==================== СИСТЕМА СКАЧИВАНИЯ ====================
class DataDownloader:
    HPO_URLS = {
        'phenotype_to_genes': 'http://purl.obolibrary.org/obo/hp/hpoa/phenotype_to_genes.txt',
        'genes_to_phenotype': 'http://purl.obolibrary.org/obo/hp/hpoa/genes_to_phenotype.txt',
        'hpo_ontology': 'http://purl.obolibrary.org/obo/hp.obo'
    }
    CLINVAR_URL = 'https://ftp.ncbi.nlm.nih.gov/pub/clinvar/tab_delimited/variant_summary.txt.gz'
    
    def __init__(self, data_dir: str = "./ciliopathy_data"):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(exist_ok=True)
        
    def download_file(self, url: str, filename: str) -> Path:
        filepath = self.data_dir / filename
        if filepath.exists():
            print(f"📁 {filename} уже существует, пропускаем")
            return filepath
        print(f"📥 Скачивание {filename}...")
        try:
            response = requests.get(url, stream=True, timeout=60)
            response.raise_for_status()
            total_size = int(response.headers.get('content-length', 0))
            with open(filepath, 'wb') as f:
                with tqdm(total=total_size, unit='B', unit_scale=True, desc=filename) as pbar:
                    for chunk in response.iter_content(chunk_size=8192):
                        if chunk:
                            f.write(chunk)
                            pbar.update(len(chunk))
            print(f"✅ {filename} скачан")
            return filepath
        except Exception as e:
            print(f"❌ Ошибка {filename}: {e}")
            raise

    def download_hpo_data(self) -> Dict[str, Path]:
        print("\n" + "="*60)
        print("📥 СКАЧИВАНИЕ HPO")
        print("="*60)
        hpo_files = {}
        for key, url in self.HPO_URLS.items():
            filename = f"hpo_{key}.txt" if key != 'hpo_ontology' else 'hpo.obo'
            hpo_files[key] = self.download_file(url, filename)
        return hpo_files
    
    def download_clinvar_data(self) -> Path:
        print("\n" + "="*60)
        print("📥 СКАЧИВАНИЕ CLINVAR")
        print("="*60)
        gz_file = self.download_file(self.CLINVAR_URL, "variant_summary.txt.gz")
        txt_file = self.data_dir / "variant_summary.txt"
        if not txt_file.exists():
            print("📦 Распаковка...")
            with gzip.open(gz_file, 'rb') as f_in:
                with open(txt_file, 'wb') as f_out:
                    shutil.copyfileobj(f_in, f_out)
            print("✅ Распакован")
        return txt_file
    
    def download_all(self) -> Tuple[Dict[str, Path], Path]:
        return self.download_hpo_data(), self.download_clinvar_data()

    def parse_phenotype_to_genes(self) -> Dict[str, List[HPOData]]:
        filepath = self.hpo_files['phenotype_to_genes']
        print(f"📖 Чтение {filepath.name}...")
        with open(filepath, 'r', encoding='utf-8') as f:
            for line in tqdm(f, desc="Парсинг phenotype_to_genes"):
                if line.startswith('#'):
                    continue
                parts = line.strip().split('\t')
                if len(parts) < 4:
                    continue
                if parts[0].startswith('HP:'):
                    hpo_id, hpo_name, gene_symbol, frequency = parts[0], parts[1], parts[2], parts[3] if len(parts)>3 else ""
                else:
                    if len(parts) >= 5 and parts[3].startswith('HP:'):
                        gene_symbol, hpo_name, hpo_id, frequency = parts[1], parts[2], parts[3], parts[4] if len(parts)>4 else ""
                    else:
                        continue
                existing = {h.hpo_id for h in self.gene_to_hpo.get(gene_symbol, [])}
                if hpo_id not in existing:
                    hpo_data = HPOData(
                        hpo_id=hpo_id,
                        hpo_name=hpo_name,
                        gene_symbol=gene_symbol,
                        frequency=frequency
                    )
                    self.gene_to_hpo[gene_symbol].append(hpo_data)
        print(f"✅ Загружено {sum(len(v) for v in self.gene_to_hpo.values())} HPO ассоциаций для {len(self.gene_to_hpo)} генов")
        return self.gene_to_hpo
    
    def parse_genes_to_phenotype(self) -> Dict[str, List[HPOData]]:
        filepath = self.hpo_files.get('genes_to_phenotype')
        if not filepath or not filepath.exists():
            return self.gene_to_hpo
        print(f"📖 Чтение {filepath.name}...")
        with open(filepath, 'r', encoding='utf-8') as f:
            for line in tqdm(f, desc="Парсинг genes_to_phenotype"):
                if line.startswith('#'):
                    continue
                parts = line.strip().split('\t')
                if len(parts) >= 4:
                    gene_symbol, hpo_id, hpo_name = parts[1], parts[2], parts[3]
                    existing = {h.hpo_id for h in self.gene_to_hpo.get(gene_symbol, [])}
                    if hpo_id not in existing:
                        hpo_data = HPOData(
                            hpo_id=hpo_id,
                            hpo_name=hpo_name,
                            gene_symbol=gene_symbol
                        )
                        self.gene_to_hpo[gene_symbol].append(hpo_data)
        return self.gene_to_hpo
    
    def parse_hpo_ontology(self) -> Dict[str, str]:
        filepath = self.hpo_files.get('hpo_ontology')
        if not filepath or not filepath.exists():
            return {}
        print(f"📖 Чтение онтологии {filepath.name}...")
        hpo_desc = {}
        with open(filepath, 'r', encoding='utf-8') as f:
            current_id = ""
            current_name = ""
            for line in f:
                line = line.strip()
                if line.startswith('id:'):
                    current_id = line[3:].strip()
                elif line.startswith('name:'):
                    current_name = line[5:].strip()
                    if current_id and current_name:
                        hpo_desc[current_id] = current_name
                        current_id = ""
                        current_name = ""
        print(f"✅ Загружено {len(hpo_desc)} описаний HPO")
        return hpo_desc
    
    def get_gene_hpo_data(self, gene_symbol: str) -> List[HPOData]:
        gene_upper = gene_symbol.upper()
        if gene_upper in self.gene_to_hpo:
            return self.gene_to_hpo[gene_upper]
        for key in self.gene_to_hpo:
            if key.upper() == gene_upper:
                return self.gene_to_hpo[key]
        for key in self.gene_to_hpo:
            if gene_upper in key.upper() or key.upper() in gene_upper:
                return self.gene_to_hpo[key]
        return []
    
    def analyze_phenotype_categories(self, hpo_list: List[HPOData]) -> Dict[str, int]:
        categories = {cat: 0 for cat in self.PHENOTYPE_CATEGORIES}
        for hpo in hpo_list:
            hpo_id = hpo.hpo_id
            matched = False
            for cat, info in self.PHENOTYPE_CATEGORIES.items():
                if hpo_id in info.get('ids', []):
                    categories[cat] += 1
                    matched = True
                    break
            if not matched:
                hpo_name_lower = hpo.hpo_name.lower()
                for cat, info in self.PHENOTYPE_CATEGORIES.items():
                    for kw in info.get('keywords', []):
                        if kw in hpo_name_lower:
                            categories[cat] += 1
                            matched = True
                            break
                    if matched:
                        break
        return categories
    
    def calculate_phenotype_score(self, hpo_list: List[HPOData]) -> float:
        if not hpo_list:
            return 0.0
        total = sum(h.frequency_score for h in hpo_list)
        max_possible = len(hpo_list) * 3.0
        return round((total / max_possible) * 10, 2) if max_possible > 0 else 0.0

# ==================== ПАРСЕР CLINVAR ====================
class ClinVarParser:
    def __init__(self, clinvar_file: Path):
        self.clinvar_file = clinvar_file
        self.gene_to_variants = defaultdict(list)
        self.synonyms = {
            'BBS3': 'ARL6', 'ARL6': 'BBS3',
            'BBS6': 'MKKS', 'MKKS': 'BBS6',
            'NPHP2': 'INVS', 'INVS': 'NPHP2',
            'IFT144': 'WDR19', 'WDR19': 'IFT144',
            'IFT121': 'WDR35', 'WDR35': 'IFT121',
            'CCDC11': 'CFAP53', 'CFAP53': 'CCDC11',
            'CTPP4': 'TTC12', 'TTC12': 'CTPP4',
            'BBS17': 'LZTFL1', 'LZTFL1': 'BBS17',
            'IFT54': 'TRAF3IP1', 'TRAF3IP1': 'IFT54'
        }
    
    def parse_clinvar_data(self, target_genes: List[str]) -> Dict[str, List[ClinVarVariant]]:
        print(f"📖 Парсинг ClinVar из {self.clinvar_file.name}...")
        target_set = {g.upper() for g in target_genes}
        for gene in list(target_set):
            if gene in self.synonyms:
                target_set.add(self.synonyms[gene].upper())
        
        try:
            chunksize = 100000
            total_rows = 0
            found = 0
            for chunk in tqdm(pd.read_csv(self.clinvar_file, sep='\t', dtype=str,
                                          chunksize=chunksize, low_memory=False),
                              desc="Чтение ClinVar"):
                total_rows += len(chunk)
                needed = ['GeneSymbol', 'ClinicalSignificance', 'Type', 'ReviewStatus',
                          'Chromosome', 'Start', 'AgeOfOnset', 'PhenotypeIDs']
                col_map = {}
                for col in needed:
                    for actual in chunk.columns:
                        if col.lower() in actual.lower():
                            col_map[col] = actual
                            break
                if 'GeneSymbol' not in col_map:
                    continue
                chunk['GeneSymbol_upper'] = chunk[col_map['GeneSymbol']].str.upper()
                filtered = chunk[chunk['GeneSymbol_upper'].isin(target_set)]
                for _, row in filtered.iterrows():
                    gene = str(row[col_map['GeneSymbol']]).strip().upper()
                    if gene in self.synonyms:
                        gene = self.synonyms[gene]
                    variant_id = str(row.get(col_map.get('VariationID', 'VariationID'), '')).strip()
                    clin_sig = str(row.get(col_map.get('ClinicalSignificance', 'ClinicalSignificance'), '')).strip()
                    type_raw = str(row.get(col_map.get('Type', 'Type'), '')).strip().lower()
                    variant_type = self._classify_variant_type(type_raw)
                    review = str(row.get(col_map.get('ReviewStatus', 'ReviewStatus'), '')).strip()
                    chrom = str(row.get(col_map.get('Chromosome', 'Chromosome'), '')).strip()
                    pos = str(row.get(col_map.get('Start', 'Start'), '')).strip()
                    age = str(row.get(col_map.get('AgeOfOnset', 'AgeOfOnset'), '')).strip()
                    phen_ids = str(row.get(col_map.get('PhenotypeIDs', 'PhenotypeIDs'), '')).strip()
                    
                    var = ClinVarVariant(
                        variant_id=variant_id,
                        gene_symbol=gene,
                        clinical_significance=clin_sig,
                        variant_type=variant_type,
                        review_status=review,
                        chromosome=chrom,
                        position=pos,
                        age_of_onset=age,
                        phenotype_ids=phen_ids
                    )
                    self.gene_to_variants[gene].append(var)
                    found += 1
            print(f"✅ Обработано {total_rows} строк, найдено {found} вариантов для {len(self.gene_to_variants)} генов")
            if found == 0:
                print("⚠️ Не найдено вариантов, пробуем альтернативный парсинг...")
                self._alternative_parsing(target_set)
        except Exception as e:
            print(f"❌ Ошибка: {e}")
            import traceback
            traceback.print_exc()
            print("🔄 Пробуем простой парсинг...")
            self._simple_parse(target_set)
        return self.gene_to_variants
    
    def _classify_variant_type(self, type_str: str) -> str:
        if not type_str:
            return 'other'
        type_low = type_str.lower()
        if 'missense' in type_low:
            return 'missense'
        if 'nonsense' in type_low or 'stop_gain' in type_low or 'stop gain' in type_low:
            return 'nonsense'
        if 'frameshift' in type_low:
            return 'frameshift'
        if 'splice' in type_low:
            return 'splice'
        if 'deletion' in type_low:
            return 'deletion'
        if 'insertion' in type_low:
            return 'insertion'
        if 'duplication' in type_low:
            return 'duplication'
        if 'indel' in type_low:
            return 'indel'
        if 'synonymous' in type_low or 'silent' in type_low:
            return 'synonymous'
        if 'stop_loss' in type_low:
            return 'stop_loss'
        if 'start_loss' in type_low:
            return 'start_loss'
        if 'intronic' in type_low or 'intron' in type_low:
            return 'intronic'
        if 'utr' in type_low or 'untranslated' in type_low:
            return 'utr'
        if 'regulatory' in type_low:
            return 'regulatory'
        if 'intergenic' in type_low:
            return 'intergenic'
        return 'other'
    
    def _alternative_parsing(self, target_set: Set[str]):
        try:
            with open(self.clinvar_file, 'r', encoding='utf-8') as f:
                header = f.readline().strip().split('\t')
                idx = {col: i for i, col in enumerate(header)}
                gene_col = idx.get('GeneSymbol', -1)
                if gene_col == -1:
                    for i, col in enumerate(header):
                        if 'gene' in col.lower() and 'symbol' in col.lower():
                            gene_col = i
                            break
                if gene_col == -1:
                    print("❌ Колонка GeneSymbol не найдена")
                    return
                clin_col = idx.get('ClinicalSignificance', -1)
                type_col = idx.get('Type', -1)
                age_col = idx.get('AgeOfOnset', -1)
                for line in tqdm(f, desc="Альтернативный парсинг"):
                    parts = line.strip().split('\t')
                    if len(parts) <= max(gene_col, clin_col, type_col, age_col):
                        continue
                    gene = parts[gene_col].strip().upper()
                    if gene in target_set:
                        var = ClinVarVariant(
                            variant_id=parts[1] if len(parts)>1 else "",
                            gene_symbol=gene,
                            clinical_significance=parts[clin_col] if clin_col!=-1 else "",
                            variant_type=self._classify_variant_type(parts[type_col] if type_col!=-1 else ""),
                            review_status=parts[idx.get('ReviewStatus', -1)] if 'ReviewStatus' in idx else "",
                            chromosome=parts[idx.get('Chromosome', -1)] if 'Chromosome' in idx else "",
                            position=parts[idx.get('Start', -1)] if 'Start' in idx else "",
                            age_of_onset=parts[age_col] if age_col!=-1 else "",
                            phenotype_ids=parts[idx.get('PhenotypeIDs', -1)] if 'PhenotypeIDs' in idx else ""
                        )
                        self.gene_to_variants[gene].append(var)
            print(f"✅ Альтернативным методом найдено {sum(len(v) for v in self.gene_to_variants.values())} вариантов")
        except Exception as e:
            print(f"❌ Альтернативный парсинг не удался: {e}")
    
    def _simple_parse(self, target_set: Set[str]):
        print("🔄 Простой парсинг...")
        try:
            with open(self.clinvar_file, 'r', encoding='utf-8') as f:
                for line in tqdm(f, desc="Простой парсинг"):
                    if '\t' not in line:
                        continue
                    parts = line.strip().split('\t')
                    if len(parts) < 5:
                        continue
                    for i, part in enumerate(parts[:10]):
                        gene_candidate = str(part).strip().upper()
                        if gene_candidate in target_set:
                            var = ClinVarVariant(
                                variant_id=parts[1] if len(parts)>1 else "",
                                gene_symbol=gene_candidate,
                                clinical_significance=parts[5] if len(parts)>5 else "",
                                variant_type=self._classify_variant_type(parts[6] if len(parts)>6 else ""),
                                review_status=parts[7] if len(parts)>7 else "",
                                chromosome=parts[12] if len(parts)>12 else "",
                                position=parts[13] if len(parts)>13 else "",
                                age_of_onset=parts[14] if len(parts)>14 else "",
                                phenotype_ids=parts[15] if len(parts)>15 else ""
                            )
                            self.gene_to_variants[gene_candidate].append(var)
                            break
            print(f"✅ Простым парсингом найдено {sum(len(v) for v in self.gene_to_variants.values())} вариантов")
        except Exception as e:
            print(f"❌ Простой парсинг не удался: {e}")
    
    def get_gene_variants(self, gene_symbol: str) -> List[ClinVarVariant]:
        gene_upper = gene_symbol.upper()
        if gene_upper in self.gene_to_variants:
            return self.gene_to_variants[gene_upper]
        if gene_upper in self.synonyms:
            alt = self.synonyms[gene_upper].upper()
            if alt in self.gene_to_variants:
                return self.gene_to_variants[alt]
        return []
    
    def analyze_variants(self, variants: List[ClinVarVariant]) -> Dict[str, Any]:
        if not variants:
            return {
                'total': 0,
                'pathogenic': 0,
                'likely_pathogenic': 0,
                'vus': 0,
                'likely_benign': 0,
                'benign': 0,
                'conflicting': 0,
                'types': {},
                'pathogenicity_score': 0.0,
                'age_distribution': {}
            }
        pathogenic = sum(1 for v in variants if v.is_pathogenic)
        likely_pathogenic = sum(1 for v in variants if v.is_likely_pathogenic)
        vus = sum(1 for v in variants if v.is_vus)
        likely_benign = sum(1 for v in variants if v.is_likely_benign)
        benign = sum(1 for v in variants if v.is_benign)
        conflicting = sum(1 for v in variants if 'conflict' in v.clinical_significance.lower())
        type_counts = Counter(v.variant_type for v in variants)
        age_counts = Counter()
        for v in variants:
            if v.age_of_onset and v.age_of_onset.lower() not in ['', 'not provided', 'na']:
                age_counts[v.age_of_onset] += 1
        pathogenicity_score = ((pathogenic + likely_pathogenic) / len(variants)) * 10 if variants else 0.0
        return {
            'total': len(variants),
            'pathogenic': pathogenic,
            'likely_pathogenic': likely_pathogenic,
            'vus': vus,
            'likely_benign': likely_benign,
            'benign': benign,
            'conflicting': conflicting,
            'types': dict(type_counts),
            'pathogenicity_score': round(pathogenicity_score, 2),
            'age_distribution': dict(age_counts)
        }

# ==================== ИНТЕГРАТОР ====================
class CiliopathyDataIntegrator:
    def __init__(self, genes: List[str]):
        self.genes = list(set(genes))
        self.gene_summaries = {}
        self.synonym_map = {
            'ARL6': 'BBS3', 'BBS3': 'ARL6',
            'MKKS': 'BBS6', 'BBS6': 'MKKS',
            'INVS': 'NPHP2', 'NPHP2': 'INVS',
            'WDR19': 'IFT144', 'IFT144': 'WDR19',
            'WDR35': 'IFT121', 'IFT121': 'WDR35',
            'CFAP53': 'CCDC11', 'CCDC11': 'CFAP53',
            'TTC12': 'CTPP4', 'CTPP4': 'TTC12',
            'LZTFL1': 'BBS17', 'BBS17': 'LZTFL1',
            'TRAF3IP1': 'IFT54', 'IFT54': 'TRAF3IP1'
        }
    
    def integrate_data(self, hpo_parser: HPOParser, clinvar_parser: ClinVarParser):
        print("\n" + "="*60)
        print("🔄 ИНТЕГРАЦИЯ ДАННЫХ")
        print("="*60)
        for gene in tqdm(self.genes, desc="Обработка генов"):
            hpo_data = hpo_parser.get_gene_hpo_data(gene)
            if not hpo_data and gene in self.synonym_map:
                hpo_data = hpo_parser.get_gene_hpo_data(self.synonym_map[gene])
            phenotype_categories = hpo_parser.analyze_phenotype_categories(hpo_data)
            phenotype_score = hpo_parser.calculate_phenotype_score(hpo_data)
            
            variants = clinvar_parser.get_gene_variants(gene)
            variant_analysis = clinvar_parser.analyze_variants(variants)
            
            summary = GeneSummary(
                gene_symbol=gene,
                hpo_terms=hpo_data,
                hpo_count=len(hpo_data),
                phenotype_categories=phenotype_categories,
                phenotype_score=phenotype_score,
                clinvar_variants=variants,
                total_variants=variant_analysis['total'],
                pathogenic_count=variant_analysis['pathogenic'],
                likely_pathogenic_count=variant_analysis['likely_pathogenic'],
                vus_count=variant_analysis['vus'],
                likely_benign_count=variant_analysis['likely_benign'],
                benign_count=variant_analysis['benign'],
                conflicting_count=variant_analysis['conflicting'],
                variant_types=variant_analysis['types'],
                pathogenicity_score=variant_analysis['pathogenicity_score'],
                age_of_onset_distribution=variant_analysis.get('age_distribution', {})
            )
            summary.data_quality = self.calculate_data_quality(summary)
            summary.disease_association_score = self.calculate_disease_score(summary)
            self.gene_summaries[gene] = summary
        print(f"✅ Интегрировано для {len(self.gene_summaries)} генов")
        return self.gene_summaries
    
    def calculate_data_quality(self, summary: GeneSummary) -> float:
        metrics = [
            1 if summary.hpo_count > 0 else 0,
            1 if summary.total_variants > 0 else 0,
            1 if (summary.pathogenic_count + summary.likely_pathogenic_count) > 0 else 0,
            1 if summary.hpo_count > 5 and summary.total_variants > 5 else 0
        ]
        return sum(metrics) / len(metrics) * 100
    
    def calculate_disease_score(self, summary: GeneSummary) -> float:
        pheno_comp = summary.phenotype_score / 10.0
        patho_comp = summary.pathogenicity_score / 10.0 if summary.total_variants > 0 else 0.0
        evid_comp = min(1.0, (summary.hpo_count + summary.total_variants) / 100)
        score = (0.4 * pheno_comp + 0.4 * patho_comp + 0.2 * evid_comp) * 10
        return round(score, 2)
    
    def export_to_dataframe(self) -> pd.DataFrame:
        rows = []
        for gene, summary in self.gene_summaries.items():
            row = {
                'gene_symbol': gene,
                'hpo_term_count': summary.hpo_count,
                'phenotype_score': summary.phenotype_score,
                'total_variants': summary.total_variants,
                'pathogenic': summary.pathogenic_count,
                'likely_pathogenic': summary.likely_pathogenic_count,
                'vus': summary.vus_count,
                'likely_benign': summary.likely_benign_count,
                'benign': summary.benign_count,
                'conflicting': summary.conflicting_count,
                'pathogenicity_score': summary.pathogenicity_score,
                'data_quality': summary.data_quality,
                'disease_association_score': summary.disease_association_score,
                'age_of_onset_counts': str(summary.age_of_onset_distribution) if summary.age_of_onset_distribution else ''
            }
            for cat, cnt in summary.phenotype_categories.items():
                if cnt > 0:
                    row[f'pheno_{cat.lower()}'] = cnt
            for vtype, cnt in summary.variant_types.items():
                if cnt > 0:
                    row[f'var_{vtype}'] = cnt
            rows.append(row)
        df = pd.DataFrame(rows).fillna(0)
        for col in df.columns:
            if col not in ['gene_symbol', 'age_of_onset_counts']:
                try:
                    df[col] = pd.to_numeric(df[col])
                except:
                    pass
        return df.sort_values('disease_association_score', ascending=False)
    
    def export_to_json(self, filename: str = "ciliopathy_genes_data.json"):
        print(f"\n💾 Экспорт в JSON ({filename})...")
        data_dict = {}
        for gene, summary in self.gene_summaries.items():
            data_dict[gene] = summary.to_dict()
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(data_dict, f, indent=2, ensure_ascii=False, default=str)
        print(f"✅ Данные экспортированы в {filename}")

            print(f"VUS: {stats['total_vus']:,} ({stats['total_vus']/stats['total_variants']*100:.1f}%)")
        print(f"Средний фенотипический score: {stats['avg_phenotype_score']:.2f}/10")
        print(f"Средний патогенетический score: {stats['avg_pathogenicity_score']:.2f}/10")
        print(f"Среднее качество данных: {stats['avg_data_quality']:.1f}%")
        print("\n🏆 ТОП-10 ГЕНОВ ПО АССОЦИАЦИИ:")
        for item in stats['top_genes_by_disease_score']:
            print(f"  {item['gene']}: {item['score']:.1f}/10")
        return stats

        html = """

        <script>
            function filterTable() {
                var input = document.getElementById('search');
                var filter = input.value.toUpperCase();
                var table = document.getElementById('geneTable');
                var tr = table.getElementsByTagName('tr');
                for (var i = 1; i < tr.length; i++) {
                    var td = tr[i].getElementsByTagName('td')[0];
                    if (td) {
                        if (td.textContent.toUpperCase().indexOf(filter) > -1) tr[i].style.display = '';
                        else tr[i].style.display = 'none';
                    }
                }
            }
            function sortTable(n) {
                var table = document.getElementById('geneTable');
                var rows = table.rows;
                var switching = true;
                var dir = 'asc';
                while (switching) {
                    switching = false;
                    for (var i = 1; i < rows.length-1; i++) {
                        var x = rows[i].getElementsByTagName('td')[n];
                        var y = rows[i+1].getElementsByTagName('td')[n];
                        var xVal = parseFloat(x.innerHTML) || x.innerHTML;
                        var yVal = parseFloat(y.innerHTML) || y.innerHTML;
                        if ((dir === 'asc' && xVal > yVal) || (dir === 'desc' && xVal < yVal)) {
                            rows[i].parentNode.insertBefore(rows[i+1], rows[i]);
                            switching = true;
                            break;
                        }
                    }
                    if (!switching && dir === 'asc') { dir = 'desc'; switching = true; }
                }
            }
        </script>

        <body>
        <div class="header"><h1>📊 Анализ генов цилиопатий</h1><p>Дата: {date}</p></div>

        <div class="section"><h2>🧬 Данные по генам</h2><table id="geneTable"><thead><tr>
            <th onclick="sortTable(0)">Ген</th>
            <th onclick="sortTable(1)">HPO</th>
            <th onclick="sortTable(2)">PhenoScore</th>
            <th onclick="sortTable(3)">Вариантов</th>
            <th onclick="sortTable(4)">Патогенных+LP</th>
            <th onclick="sortTable(5)">VUS</th>
            <th onclick="sortTable(6)">Оценка ассоциации</th>
            <th onclick="sortTable(7)">Качество</th>
        </tr></thead><tbody>
        {rows}
        </tbody></table></div>
        </body></html>

        metrics_html = ""
        metrics = {
            'Среднее качество': f"{df['data_quality'].mean():.1f}%",
            'Средняя оценка ассоциации': f"{df['disease_association_score'].mean():.2f}/10",

            'Всего вариантов': f"{df['total_variants'].sum():,}",
            'Патогенных+LP': f"{(df['pathogenic'] + df['likely_pathogenic']).sum():,}",
            'Гены с HPO': f"{(df['hpo_term_count']>0).sum()}",
            'Гены с вариантами': f"{(df['total_variants']>0).sum()}"
        }
        for k,v in metrics.items():
            metrics_html += f'<div class="metric-box"><div class="metric-value">{v}</div><div class="metric-label">{k}</div></div>'
        
        rows_html = ""
        for _, row in df.iterrows():
            rows_html += f"<tr><td>{row['gene_symbol']}</td><td>{int(row['hpo_term_count'])}</td><td>{row['phenotype_score']:.1f}</td><td>{int(row['total_variants'])}</td><td>{int(row['pathogenic'] + row['likely_pathogenic'])}</td><td>{int(row['vus'])}</td><td>{row['disease_association_score']:.1f}</td><td>{row['data_quality']:.1f}%</td></tr>"
        
        with open('ciliopathy_report.html', 'w', encoding='utf-8') as f:
            f.write(html.format(date=datetime.now().strftime("%Y-%m-%d %H:%M"), metrics=metrics_html, rows=rows_html))
        print("✅ Отчёт сохранён в ciliopathy_report.html")

# ==================== ГЛАВНАЯ ФУНКЦИЯ ====================
def main():
    print("="*80)
    print("🚀 ЗАПУСК РАСШИРЕННОГО АНАЛИЗА")
    print("="*80)
    try:
        # Загрузка
        downloader = DataDownloader()
        hpo_files, clinvar_file = downloader.download_all()
        
        # HPO
        print("\n📖 Парсинг HPO...")
        hpo_parser = HPOParser(hpo_files)
        hpo_parser.parse_phenotype_to_genes()
        hpo_parser.parse_genes_to_phenotype()
        
        # ClinVar
        print("\n📖 Парсинг ClinVar...")
        clinvar_parser = ClinVarParser(clinvar_file)
        clinvar_parser.parse_clinvar_data(GENES)
        
        # Интеграция
        print("\n🔄 Интеграция...")
        integrator = CiliopathyDataIntegrator(GENES)
        integrator.integrate_data(hpo_parser, clinvar_parser)
        
        # Экспорт в DataFrame и файлы
        df = integrator.export_to_dataframe()
        df.to_csv('ciliopathy_genes_data.csv', index=False, encoding='utf-8-sig')
        df.to_excel('ciliopathy_genes_data.xlsx', index=False, engine='openpyxl')
        integrator.export_to_json('ciliopathy_genes_data.json')
        
        # Статистика
        stats = integrator.generate_statistics_report()
        
        # Визуализация
        visualizer = DataVisualizer()
        visualizer.plot_summary_charts(df, stats)
        visualizer.create_interactive_report(df)
        
        # ZIP-архив
        files_to_zip = ['ciliopathy_genes_data.xlsx', 'ciliopathy_genes_data.csv',
                        'ciliopathy_genes_data.json', 'ciliopathy_analysis_summary.png',
                        'ciliopathy_report.html']
        existing = [f for f in files_to_zip if Path(f).exists()]
        if existing:
            zip_name = 'ciliopathy_analysis_results.zip'
            with zipfile.ZipFile(zip_name, 'w') as zf:
                for f in existing:
                    zf.write(f)
            print(f"✅ ZIP-архив создан: {zip_name}")
            try:
                from google.colab import files
                files.download(zip_name)
            except ImportError:
                from IPython.display import FileLink
                display(FileLink(zip_name))
        
        print("\n🎉 ВСЕ ДАННЫЕ СОБРАНЫ!")
        print("📁 Файлы: ciliopathy_genes_data.xlsx, .csv, .json, .png, .html, .zip")
        return integrator, df
    except Exception as e:
        print(f"❌ КРИТИЧЕСКАЯ ОШИБКА: {e}")
        import traceback
        traceback.print_exc()
        return None, None

if __name__ == "__main__":
    main()
